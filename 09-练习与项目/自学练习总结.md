## 一、变量与运算符（题目 1–10）

### 知识点

| 类型 | 说明 | 示例 |
|------|------|------|
| `int` | 整数类型 | `int age = 20;` |
| `float` | 单精度浮点，需加 `f` 后缀 | `float price = 12.3f;` |
| `double` | 双精度浮点 | `double total = price + quantity;` |
| 输入转换 | `Convert.ToInt32()` / `double.Parse()` | `int num = Convert.ToInt32(Console.ReadLine());` |
| 三目运算符 | `条件 ? 值1 : 值2` | `bool isRaining = bool01 == "是" ? true : false;` |
| 取模 | `%` | `num % 2` 判断奇偶 |
| 交换变量 | 加减法交换 | `a += b; b = a - b; a -= b;` |

## 四、方法（函数）（题目 31–38）

### 知识点

| 概念 | 说明 |
|------|------|
| 方法定义 | `public static 返回类型 方法名(参数列表)` |
| `void` | 无返回值 |
| 返回值 | `return` 语句 |
| 方法重载 | 同名方法，参数列表不同 |
| 访问修饰符 | `public` / `internal` / `private` |
| XML 注释 | `/// <summary>` 生成文档注释 |

### 代码

```csharp
using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;
using System.Threading.Tasks;

namespace 四_方法_函数_
{
    internal class Program
    {
        /// &lt;summary&gt;41. 无参无返回值，输出 Hello&lt;/summary&gt;
        public static void SayHello()
        {
            Console.WriteLine("Hello, C#!");
        }

        /// &lt;summary&gt;32. 返回两个整数中的最大值&lt;/summary&gt;
        public static int GetMax(int num1, int num2)
        {
            if (num1 &gt; num2) return num1;
            else return num2;
        }

        /// &lt;summary&gt;33. 判断是否为偶数&lt;/summary&gt;
        public static bool IsEven(int num)
        {
            return (num % 2 == 0);
        }

        /// &lt;summary&gt;34. 打印 n 行 * 正方形&lt;/summary&gt;
        public static void PrintStar(int n)
        {
            for (int i = 1; i &lt;= n; i++)
            {
                for (int j = 1; j &lt;= n; j++) { Console.Write("* "); }
                Console.WriteLine();
            }
        }

        /// &lt;summary&gt;36. 计算 n 的阶乘&lt;/summary&gt;
        public static long Factorial(int n)
        {
            long mul = 1;
            for (long i = 1; i &lt;= n; i++) { mul *= i; }
            return mul;
        }

        /// &lt;summary&gt;36. 计算数组所有元素之和&lt;/summary&gt;
        public static int SumArray(int[] arr)
        {
            int sum = 0;
            for (int i = 0; i &lt; arr.Length; i++) { sum += arr[i]; }
            return sum;
        }

        /// &lt;summary&gt;37. 反转字符串&lt;/summary&gt;
        public static string ReverseString(string str)
        {
            char[] chars = new char[str.Length];
            for (int i = 0; i &lt; str.Length; i++)
            {
                chars[str.Length - i - 1] = str[i];
            }
            return new string(chars);
        }

        public static int Add(int a, int b) { return a + b; }

        /// &lt;summary&gt;38. 方法重载：三个整数相加&lt;/summary&gt;
        public static int Add(int a, int b, int c) { return a + b + c; }

        static void Main(string[] args)
        {
            SayHello();
            int a = 123, b = 1234;
            Console.WriteLine("最大值：" + GetMax(a, b));
            Console.WriteLine(IsEven(45));
            PrintStar(9);
            Console.WriteLine(Factorial(10));
            int[] ar = { 1, 2, 3, 4, 5, 6 };
            Console.WriteLine(SumArray(ar));
            string strings = "123456";
            Console.WriteLine(strings);
            Console.WriteLine(Add(1, 2));
            Console.WriteLine(Add(1, 2, 3));
        }
    }
}
```

## 五、随机数（题目 39–42）

### 知识点

- `Random rnd = new Random()` 创建随机数生成器
- `rnd.Next(min, max)` — 生成 [min, max) 范围的随机整数
- `rnd.Next(0, 10)` — 生成 0–9（含 0，不含 10）

### 代码

```csharp
using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;
using System.Threading.Tasks;

namespace 五_随机数
{
    internal class Program
    {
        static void Main(string[] args)
        {
            Random rnd = new Random();

            // 39. 生成 10 个 1-100 的随机整数
            for (int i = 0; i < 10; i++)
            {
                int r = rnd.Next(1, 100);
                Console.WriteLine(r);
            }

            // 40. 生成 4 位数字验证码
            int[] VerCode = new int[4];
            for (int i = 0; i < VerCode.Length; i++)
            {
                VerCode[i] = rnd.Next(0, 10);
            }
            Console.WriteLine("验证码：");
            foreach (int i in VerCode) { Console.Write(i); }
            Console.WriteLine();

            // 41. 模拟掷骰子 1000 次，统计各点数
            int a = 1, b = 2, c = 3, d = 4, e = 5, f = 6;
            for (int i = 0; i < 1000; i++)
            {
                int six = rnd.Next(1, 7);
                if (six == 1) { a++; }
                else if (six == 2) { b++; }
                else if (six == 3) { c++; }
                else if (six == 4) { d++; }
                else if (six == 5) { e++; }
                else if (six == 6) { f++; }
            }
            Console.WriteLine(
                $"1点出现次数：{a}\n2点出现次数：{b}\n3点出现次数：{c}\n" +
                $"4点出现次数：{d}\n5点出现次数：{e}\n6点出现次数：{f}");

            // 42. 随机抽取水果
            string[] fruit = { "苹果", "香蕉", "橘子", "葡萄" };
            for (int i = 0; i < fruit.Length; i++)
            {
                int r1 = rnd.Next(0, 4);
                (fruit[i], fruit[r1]) = (fruit[r1], fruit[i]);
            }
            Console.WriteLine(fruit[1]);
        }
    }
}
```

> ⚠️ **第 42 题 bug**：通过交换数组元素再取 `fruit[1]` 来随机取值，逻辑有误。正确写法：`fruit[rnd.Next(0, fruit.Length)]`。

## 六、一维数组（题目 43–50）

### 知识点

```csharp

            }

            int num = int.Parse(Console.ReadLine());

            // 49. 统计正数、负数、零的个数
            int[] ARR = { -11, 23, -2, -3, 4, 5, 6, 78, 0 };
            int number1 = 0, number2 = 0, number3 = 0;
            for (int i = 0; i &lt; ARR.Length; i++)
            {
                if (ARR[i] == 0) { number1++; }
                else if (ARR[i] &gt; 0) { number2++; }
                else { number3++; }
            }
            Console.WriteLine($"正数有：{number2}个，负数有：{number3}个，零有：{number1}个");

            // 50. 两个等长数组对应元素相加
            int[] one = { 1, 2, 3, 4, 5, 6, 7, 8 };
            int[] two = { 9, 78, 56, 565, 44, 32, 1, 2 };
            int[] three = new int[one.Length];
            for (int i = 0; i &lt; one.Length; i++) { three[i] = one[i] + two[i]; }
            foreach (int i in three) { Console.WriteLine(i); }
        }
    }
```

### 常见错误提醒

1. **数组越界**：`for` 循环条件应为 `i < arr.Length`，`<=` 会导致越界。
2. **变量命名不规范**：如 `hudred`（应为 `hundred`）、`bool01` 等。
3. **逻辑判断疏漏**：如第 25 题 if/else 输出相同、第 44 题最小值判断条件错误。
4. **switch 缺少 break**：第 14 题 default 分支缺少 `break`，可能 fall-through。
5. **随机取值逻辑错误**：第 42 题应直接使用 `rnd.Next(0, length)` 取索引。
